cmake_minimum_required(VERSION 3.10)
project(SSHWarden C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_executable(SSHWarden main.c)

target_compile_options(SSHWarden PRIVATE
    -Wall
    -Wextra
    -Werror
    -pedantic
)

target_compile_definitions(SSHWarden PRIVATE
    _POSIX_C_SOURCE=200809L
)

install(TARGETS SSHWarden
    RUNTIME DESTINATION bin
)
